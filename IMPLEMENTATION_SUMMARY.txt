🚀 QUICK START (3 STEPS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  CONFIGURE DATABASE (2 min)
    Edit .env.local and add:
    DATABASE_URL="your-mongodb-connection-string"
    ADMIN_PASSWORD="your-secure-password"

2️⃣  INITIALIZE DATABASE (2 min)
    bunx prisma migrate dev --name init
    bunx prisma studio
    (Create 3-5 categories in the UI)

3️⃣  START USING (1 min)
    bun dev
    Visit: http://localhost:3000/blog
    Visit: http://localhost:3000/admin/blogs

📝 CREATE YOUR FIRST BLOG
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to http://localhost:3000/admin/blogs
2. Login with your admin password
3. Click "+ Create New Blog"
4. Fill in:
   • Title: Your blog title
   • Category: Select from dropdown
   • Excerpt: Short summary
   • Content: Write in markdown
   • Image: Upload featured image
   • Author: Your name
   • Publish: Check to make it public
5. Click "Create Blog"
6. Visit /blog to see it live!
